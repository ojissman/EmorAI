    26→export type AnthropicErrorType =
    27→  | 'invalid_request_error'
    28→  | 'authentication_error'
    29→  | 'permission_error'
    30→  | 'not_found_error'
    31→  | 'rate_limit_error'
    32→  | 'api_error'
    33→  | 'overloaded_error';
    34→
    35→export function anthropicError(
    36→  status: number,
    37→  type: AnthropicErrorType,
    38→  message: string,
    39→  headers?: Record<string, string>
    40→): Response {
    41→  return corsJson({ type: 'error', error: { type, message } }, { status, headers });
    42→}
    43→
    44→/** Shapes a protocol-agnostic auth failure into the Anthropic envelope. */
    45→export function anthropicAuthFailure(failure: AuthFailure): Response {
    46→  const headers: Record<string, string> = {};
    47→  if (failure.retryAfterSec) headers['retry-after'] = String(failure.retryAfterSec);
    48→  return anthropicError(failure.status, failure.anthropicType, failure.message, headers);
    49→}
    50→
    51→/** Maps a provider (upstream) failure into the Anthropic error envelope. */
    52→export function anthropicProviderError(code: string, message: string, status: number): Response {
    53→  const type: AnthropicErrorType =
    54→    code === 'rate_limited'
    55→      ? 'rate_limit_error'
    56→      : code === 'bad_request'
    57→        ? 'invalid_request_error'
    58→        : code === 'model_unavailable'
    59→          ? 'api_error'
    60→          : status >= 500
    61→            ? 'overloaded_error'
    62→            : 'api_error';
    63→  const headers: Record<string, string> = {};
    64→  if (code === 'rate_limited') headers['retry-after'] = '15';
    65→  return anthropicError(status === 429 ? 429 : status >= 500 ? (status === 502 ? 529 : status) : status, type, message, headers);
    66→}

   176→type RawBlock = AnthropicContentBlock;
   177→
   178→function joinText(blocks: RawBlock[]): string {
   179→  return blocks
   180→    .filter((b): b is Extract<RawBlock, { type: 'text' }> => b.type === 'text')
   181→    .map((b) => b.text)
   182→    .join('\n');
   183→}
   184→
   185→function toolResultToText(block: Extract<RawBlock, { type: 'tool_result' }>): string {
   186→  let text: string;
   187→  if (typeof block.content === 'string') {
   188→    text = block.content;
   189→  } else if (Array.isArray(block.content)) {
   190→    const parts = block.content.map((p) => {
   191→      if (typeof (p as { text?: unknown }).text === 'string') return (p as { text: string }).text;
   192→      if ((p as { type?: string }).type === 'image') {
   193→        throw new ProtocolTranslationError(
   194→          'Images inside tool_result blocks are not supported by the underlying provider. Return tool results as text.',
   195→          400
   196→        );
   197→      }
   198→      return '';
   199→    });
   200→    text = parts.filter(Boolean).join('\n');
   201→  } else {
   202→    text = '';
   203→  }
   204→  if (block.is_error) {
   205→    text = text.length > 0 ? `[tool error] ${text}` : '[tool error]';
   206→  }
   207→  // The provider expects tool content as a string (JSON or plain text).
   208→  return text.length > 0 ? text : '(empty tool result)';
   209→}
   210→
   211→function imageSourceToPart(block: Extract<RawBlock, { type: 'image' }>): ProviderContentPart {
   212→  const source = block.source as { type?: string; url?: string; media_type?: string; data?: string } | undefined;
   213→  if (source?.type === 'url' && typeof source.url === 'string') {
   214→    return { type: 'image_url', image_url: { url: source.url } };
   215→  }
   216→  if (source?.type === 'base64' && typeof source.data === 'string') {
   217→    // Anthropic base64 image → OpenAI-style data URI
   218→    const mediaType = source.media_type || 'image/png';
   219→    return { type: 'image_url', image_url: { url: `data:${mediaType};base64,${source.data}` } };
   220→  }
   221→  // Malformed image block (passed the liberal fallback branch) — degrade to text
   222→  return { type: 'text', text: '[invalid image block]' };
   223→}
   224→
   225→function documentToText(block: Extract<RawBlock, { type: 'document' }>): string {
   226→  const source = block.source as { type?: string; content?: string; media_type?: string } | undefined;
   227→  if (source?.type === 'text' && typeof source.content === 'string') {
   228→    return source.content;
   229→  }
   230→  throw new ProtocolTranslationError(
   231→    'PDF document blocks are not supported by the underlying provider. Send document text inline or use a vision model with images.',
   232→    400
   233→  );
   234→}
   235→
   236→/** Text of a message whose content is a string or an Anthropic block array. */
   237→function contentToText(content: string | AnthropicContentBlock[]): string {
   238→  if (typeof content === 'string') return content;
   239→  return joinText(content);
   240→}
   241→
   242→/**
   243→ * Translates one normalized user/assistant message (string or block array)
   244→ * into one or more provider messages. Tool results become individual `tool`
   245→ * messages; text and images stay in `user`/`assistant` messages; thinking
   246→ * blocks are accepted and dropped (the provider does not replay reasoning in
   247→ * history).
   248→ */
   249→function translateMessage(msg: { role: 'user' | 'assistant'; content: string | AnthropicContentBlock[] }): ProviderMessage[] {
   250→  if (typeof msg.content === 'string') {
   251→    return [{ role: msg.role, content: msg.content }];
   252→  }
   253→  const blocks = msg.content;
   254→
   255→  if (msg.role === 'assistant') {
   256→    const text = joinText(blocks);
   257→    const toolUses = blocks.filter((b): b is Extract<RawBlock, { type: 'tool_use' }> => b.type === 'tool_use');
   258→    if (toolUses.length > 0) {
   259→      const toolCalls: ProviderToolCall[] = toolUses.map((u) => ({
   260→        id: u.id,
   261→        type: 'function',
   262→        function: {
   263→          name: u.name,
   264→          arguments: JSON.stringify(u.input ?? {}),
   265→        },
   266→      }));
   267→      return [{ role: 'assistant', content: text, tool_calls: toolCalls }];
   268→    }
   269→    return [{ role: 'assistant', content: text }];
   270→  }
   271→
   272→  // user message: tool_result blocks must be answered first, then any text /
   273→  // image content follows as a regular user message (ordering preserved).
   274→  const out: ProviderMessage[] = [];
   275→  for (const block of blocks) {
   276→    if (block.type === 'tool_result') {
   277→      out.push({ role: 'tool', content: toolResultToText(block), tool_call_id: block.tool_use_id });
   278→    }
   279→  }
   280→  const rest = blocks.filter((b) => b.type !== 'tool_result');
   281→  if (rest.length > 0) {
   282→    const hasImages = rest.some((b) => b.type === 'image');
   283→    if (hasImages) {
   284→      const parts: ProviderContentPart[] = rest.map((b) => {
   285→        if (b.type === 'text') return { type: 'text' as const, text: b.text };
   286→        if (b.type === 'image') return imageSourceToPart(b);
   287→        if (b.type === 'document') return { type: 'text' as const, text: documentToText(b) };
   288→        // thinking blocks in user history are dropped
   289→        return null;
   290→      });
   291→      const compact = parts.filter((p): p is ProviderContentPart => p !== null);
   292→      if (compact.length > 0) out.push({ role: 'user', content: compact });
   293→    } else {
   294→      const texts = rest
   295→        .map((b) => {
   296→          if (b.type === 'text') return b.text;
   297→          if (b.type === 'document') return documentToText(b);
   298→          return null;
   299→        })
   300→        .filter((t): t is string => t !== null);
   301→      if (texts.length > 0) out.push({ role: 'user', content: texts.join('\n') });
   302→    }
   303→  }
   304→  return out;
   305→}
   306→
   307→function translateSystem(system: AnthropicBody['system']): string | null {
   308→  if (system === undefined) return null;
   309→  if (typeof system === 'string') return system.length > 0 ? system : null;
   310→  const texts = system
   311→    .map((block) => (typeof block.text === 'string' ? block.text : ''))
   312→    .filter(Boolean);
   313→  return texts.length > 0 ? texts.join('\n\n') : null;
   314→}
   315→
   316→function translateTools(tools: AnthropicBody['tools']): ProviderTool[] | undefined {
   317→  if (!tools || tools.length === 0) return undefined;
   318→  return tools.map((t) => ({
   319→    type: 'function' as const,
   320→    function: {
   321→      name: t.name,
   322→      description: t.description ?? '',
   323→      parameters: t.input_schema ?? { type: 'object', properties: {} },
   324→    },
   325→  }));
   326→}
   327→
   328→function translateToolChoice(choice: AnthropicBody['tool_choice']): unknown {
   329→  if (!choice) return undefined;
   330→  switch (choice.type) {
   331→    case 'auto':
   332→      return 'auto';
   333→    case 'any':
   334→      return 'required';
   335→    case 'tool':
   336→      return { type: 'function', function: { name: choice.name } };
   337→  }
   338→}

   352→/** Full request translation, applying EmorAI model limits. */
   353→export function anthropicToProvider(
   354→  body: AnthropicBody,
   355→  opts: { defaultMaxTokens: number; modelMaxOutputTokens: number | null }
   356→): TranslatedRequest {
   357→  if (body.messages.length === 0) {
   358→    throw new ProtocolTranslationError('messages: at least one message is required.');
   359→  }
   360→
   361→  // Liberal role handling: normalize each message role first, then translate.
   362→  // system/developer roles merge into the system prompt; tool/function roles
   363→  // become provider tool messages; OpenAI-style tool_calls on assistant
   364→  // messages are honored. finalizeProviderMessages() then enforces the
   365→  // upstream invariants (single leading system message, no consecutive
   366→  // same-role messages, no empty messages).
   367→  const providerMessages: ProviderMessage[] = [];
   368→  const systemParts: string[] = [];
   369→  const topSystem = translateSystem(body.system);
   370→  if (topSystem) systemParts.push(topSystem);
   371→
   372→  for (const msg of body.messages) {
   373→    const role = normalizeChatRole(msg.role);
   374→    if (role === 'system') {
   375→      const text = contentToText(msg.content);
   376→      if (text.trim().length > 0) systemParts.push(text);
   377→      continue;
   378→    }
   379→    if (role === 'tool') {
   380→      // OpenAI-style tool result leaked into the Anthropic messages array
   381→      const text = contentToText(msg.content);
   382→      providerMessages.push({
   383→        role: 'tool',
   384→        content: text.trim().length > 0 ? text : '(empty tool result)',
   385→        tool_call_id: msg.tool_call_id ?? msg.name ?? `call_${providerMessages.length}`,
   386→      });
   387→      continue;
   388→    }
   389→    if (role === 'assistant' && msg.tool_calls && msg.tool_calls.length > 0) {
   390→      // OpenAI-style assistant tool_calls leak
   391→      providerMessages.push({
   392→        role: 'assistant',
   393→        content: contentToText(msg.content),
   394→        tool_calls: msg.tool_calls.map((tc, i) => ({
   395→          id: tc.id && tc.id.length > 0 ? tc.id : `call_${Date.now().toString(36)}_${i}`,
   396→          type: 'function' as const,
   397→          function: { name: tc.function.name, arguments: tc.function.arguments ?? '{}' },
   398→        })),
   399→      });
   400→      continue;
   401→    }
   402→    providerMessages.push(...translateMessage({ role, content: msg.content }));
   403→  }
   404→
   405→  const messages = finalizeProviderMessages(
   406→    systemParts.length > 0 ? [{ role: 'system', content: systemParts.join('\n\n') }, ...providerMessages] : providerMessages
   407→  );
   408→  if (messages.length === 0 || messages.every((m) => m.role === 'system')) {
   409→    throw new ProtocolTranslationError('messages: the request contains no user or assistant content.');
   410→  }
   411→
   412→  // Anthropic marks max_tokens optional in newer drafts but the classic API
   413→  // requires it; we default to the model limit (friendlier than erroring).
   414→  const requestedMax = body.max_tokens ?? opts.defaultMaxTokens;
   415→  const maxTokens = Math.min(requestedMax, opts.modelMaxOutputTokens ?? requestedMax);
   416→
   417→  return {
   418→    messages,
   419→    tools: translateTools(body.tools),
   420→    toolChoice: translateToolChoice(body.tool_choice),
   421→    maxTokens,
   422→    temperature: body.temperature,
   423→    topP: body.top_p,
   424→    topK: body.top_k,
   425→    stop: body.stop_sequences && body.stop_sequences.length > 0 ? body.stop_sequences : undefined,
   426→    thinkingEnabled: body.thinking?.type === 'enabled',
   427→  };
   428→}

   432→export function estimateInputTokens(messages: ProviderMessage[], system?: string | null): number {
   433→  let chars = system?.length ?? 0;
   434→  for (const m of messages) {
   435→    if (typeof m.content === 'string') chars += m.content.length;
   436→    else if (Array.isArray(m.content)) {
   437→      for (const p of m.content) chars += p.text?.length ?? 0;
   438→    }
   439→    if (m.tool_calls) {
   440→      for (const tc of m.tool_calls) chars += tc.function.arguments.length + tc.function.name.length;
   441→    }
   442→  }
   443→  return Math.max(1, Math.ceil(chars / 4));
   444→}

   448→export type AnthropicStopReason = 'end_turn' | 'tool_use' | 'max_tokens' | 'stop_sequence' | 'refusal';
   449→
   450→export function toAnthropicStopReason(reason: string | null): AnthropicStopReason {
   451→  switch (reason) {
   452→    case 'tool_calls':
   453→    case 'function_call':
   454→      return 'tool_use';
   455→    case 'length':
   456→    case 'max_tokens':
   457→      return 'max_tokens';
   458→    case 'content_filter':
   459→      return 'refusal';
   460→    default:
   461→      return 'end_turn';
   462→  }
   463→}

   465→export type AnthropicContentOut =
   466→  | { type: 'text'; text: string }
   467→  | { type: 'tool_use'; id: string; name: string; input: Record<string, unknown> };
   468→
   469→function safeParseArguments(args: string): Record<string, unknown> {
   470→  try {
   471→    const parsed = JSON.parse(args || '{}');
   472→    return typeof parsed === 'object' && parsed !== null && !Array.isArray(parsed)
   473→      ? (parsed as Record<string, unknown>)
   474→      : { value: parsed };
   475→  } catch {
   476→    // Malformed arguments from the provider — surface raw text honestly.
   477→    return { _raw: args };
   478→  }
   479→}

   481→/** Builds the non-streaming Anthropic message response. */
   482→export function buildAnthropicMessage(params: {
   483→  id: string;
   484→  requestedModel: string;
   485→  servedModel: string | null;
   486→  content: string;
   487→  toolCalls: ProviderToolCall[];
   488→  finishReason: string | null;
   489→  usage: TokenUsage | null;
   490→}): Record<string, unknown> {
   491→  const content: AnthropicContentOut[] = [];
   492→  if (params.content && params.content.length > 0) {
   493→    content.push({ type: 'text', text: params.content });
   494→  }
   495→  for (const tc of params.toolCalls) {
   496→    content.push({
   497→      type: 'tool_use',
   498→      id: tc.id,
   499→      name: tc.function.name,
   500→      input: safeParseArguments(tc.function.arguments),
   501→    });
   502→  }
   503→  if (content.length === 0) {
   504→    content.push({ type: 'text', text: '' });
   505→  }
   506→  return {
   507→    id: params.id,
   508→    type: 'message',
   509→    role: 'assistant',
   510→    model: params.requestedModel,
   511→    content,
   512→    stop_reason: toAnthropicStopReason(params.finishReason),
   513→    stop_sequence: null,
   514→    usage: {
   515→      input_tokens: params.usage?.promptTokens ?? 0,
   516→      output_tokens: params.usage?.completionTokens ?? 0,
   517→    },
   518→    // EmorAI transparency extension (ignored by Anthropic SDKs)
   519→    emorai_served_model: params.servedModel ?? params.requestedModel,
   520→  };
   521→}
   522→
   523→// ─── Streaming translation: provider events → Anthropic SSE frames ───────────
   524→
   525→function sseFrame(event: string, data: unknown): string {
   526→  return `event: ${event}\ndata: ${JSON.stringify(data)}\n\n`;
   527→}
   528→
   529→type OpenBlock =
   530→  | { kind: 'text'; index: number }
   531→  | { kind: 'tool_use'; index: number; id: string; name: string };
   532→
   533→/**
   534→ * Stateful translator that consumes normalized provider stream events and
   535→ * produces the Anthropic SSE event sequence. Faithful to the protocol:
   536→ *   message_start → ping → (content_block_start → deltas → content_block_stop)*
   537→ *   → message_delta(stop_reason, usage) → message_stop
   538→ * Tool-call arguments stream as input_json_delta chunks.
   539→ */
   540→export class AnthropicStreamTranslator {
   541→  private readonly messageId: string;
   542→  private readonly requestedModel: string;
   543→  private blockCount = 0;
   544→  private open: OpenBlock | null = null;
   545→  private outputChars = 0;
   546→  private usage: TokenUsage | null = null;
   547→  private inputTokens: number;
   548→  private closed = false;
   549→
   550→  constructor(params: { messageId: string; requestedModel: string; estimatedInputTokens: number }) {
   551→    this.messageId = params.messageId;
   552→    this.requestedModel = params.requestedModel;
   553→    this.inputTokens = params.estimatedInputTokens;
   554→  }
   555→
   556→  private nextIndex(): number {
   557→    return this.blockCount++;
   558→  }
   559→
   560→  private closeOpenBlock(): string | null {
   561→    if (!this.open) return null;
   562→    const frame = sseFrame('content_block_stop', { type: 'content_block_stop', index: this.open.index });
   563→    this.open = null;
   564→    return frame;
   565→  }
   566→
   567→  /** Initial frames: message_start + ping. */
   568→  start(): string {
   569→    return (
   570→      sseFrame('message_start', {
   571→        type: 'message_start',
   572→        message: {
   573→          id: this.messageId,
   574→          type: 'message',
   575→          role: 'assistant',
   576→          model: this.requestedModel,
   577→          content: [],
   578→          stop_reason: null,
   579→          stop_sequence: null,
   580→          usage: { input_tokens: this.inputTokens, output_tokens: 1 },
   581→        },
   582→      }) + sseFrame('ping', { type: 'ping' })
   583→    );
   584→  }
   585→
   586→  /** Feed one normalized provider event; returns SSE frames to write. */
   587→  onEvent(ev: StreamEvent): string {
   588→    if (this.closed) return '';
   589→    switch (ev.type) {
   590→      case 'delta': {
   591→        this.outputChars += ev.content.length;
   592→        if (!this.open || this.open.kind !== 'text') {
   593→          const closeFrame = this.closeOpenBlock();
   594→          const index = this.nextIndex();
   595→          this.open = { kind: 'text', index };
   596→          const startFrame = sseFrame('content_block_start', {
   597→            type: 'content_block_start',
   598→            index,
   599→            content_block: { type: 'text', text: '' },
   600→          });
   601→          return (
   602→            (closeFrame ?? '') +
   603→            startFrame +
   604→            sseFrame('content_block_delta', {
   605→              type: 'content_block_delta',
   606→              index,
   607→              delta: { type: 'text_delta', text: ev.content },
   608→            })
   609→          );
   610→        }
   611→        return sseFrame('content_block_delta', {
   612→          type: 'content_block_delta',
   613→          index: this.open.index,
   614→          delta: { type: 'text_delta', text: ev.content },
   615→        });
   616→      }
   617→      case 'reasoning': {
   618→        // GLM reasoning trace: Anthropic surfaces extended thinking as
   619→        // thinking blocks. We stream them as thinking blocks (signature is
   620→        // empty; clients tolerate unsigned thinking on custom gateways).
   621→        if (!this.open || this.open.kind !== 'text') {
   622→          // reasoning arrives before text typically — open a thinking block
   623→          // only if nothing is open; otherwise ignore (kept simple & robust).
   624→          if (this.open) return '';
   625→          const index = this.nextIndex();
   626→          this.open = { kind: 'text', index };
   627→          // Represent reasoning as a thinking-typed content block
   628→          return (
   629→            sseFrame('content_block_start', {
   630→              type: 'content_block_start',
   631→              index,
   632→              content_block: { type: 'thinking', thinking: '', signature: '' },
   633→            }) +
   634→            sseFrame('content_block_delta', {
   635→              type: 'content_block_delta',
   636→              index,
   637→              delta: { type: 'thinking_delta', thinking: ev.content },
   638→            })
   639→          );
   640→        }
   641→        return '';
   642→      }
   643→      case 'tool-call-delta': {
   644→        return this.onToolCallDelta(ev.call);
   645→      }
   646→      case 'served-model': {
   647→        // No Anthropic frame for this; recorded by the caller.
   648→        return '';
   649→      }
   650→      case 'usage': {
   651→        this.usage = ev.usage;
   652→        return '';
   653→      }
   654→      case 'finish': {
   655→        // handled in finish(); nothing to emit now
   656→        return '';
   657→      }
   658→    }
   659→  }
   660→
   661→  private onToolCallDelta(call: ToolCallDelta): string {
   662→    let frames = '';
   663→    const sameToolOpen =
   664→      this.open && this.open.kind === 'tool_use' && this.open.id === (call.id ?? this.open.id);
   665→    if (!sameToolOpen) {
   666→      frames += this.closeOpenBlock() ?? '';
   667→      const index = this.nextIndex();
   668→      const id = call.id ?? `toolu_${Math.random().toString(36).slice(2, 12)}`;
   669→      const name = call.name ?? '';
   670→      this.open = { kind: 'tool_use', index, id, name };
   671→      frames += sseFrame('content_block_start', {
   672→        type: 'content_block_start',
   673→        index,
   674→        content_block: { type: 'tool_use', id, name, input: {} },
   675→      });
   676→    }
   677→    if (call.argumentsDelta && call.argumentsDelta.length > 0) {
   678→      this.outputChars += call.argumentsDelta.length;
   679→      frames += sseFrame('content_block_delta', {
   680→        type: 'content_block_delta',
   681→        index: this.open!.index,
   682→        delta: { type: 'input_json_delta', partial_json: call.argumentsDelta },
   683→      });
   684→    }
   685→    return frames;
   686→  }
   687→
   688→  /** Terminal frames: close block, message_delta with stop_reason + usage, message_stop. */
   689→  finish(finishReason: string | null): string {
   690→    if (this.closed) return '';
   691→    this.closed = true;
   692→    const closeFrame = this.closeOpenBlock() ?? '';
   693→    const stopReason = toAnthropicStopReason(finishReason);
   694→    const outputTokens =
   695→      this.usage?.completionTokens ?? Math.max(1, Math.ceil(this.outputChars / 4));
   696→    if (this.usage) this.inputTokens = this.usage.promptTokens;
   697→    return (
   698→      closeFrame +
   699→      sseFrame('message_delta', {
   700→        type: 'message_delta',
   701→        delta: { stop_reason: stopReason, stop_sequence: null },
   702→        usage: { output_tokens: outputTokens },
   703→      }) +
   704→      sseFrame('message_stop', { type: 'message_stop' })
   705→    );
   706→  }
   707→
   708→  /** Mid-stream error frame (Anthropic error event). */
   709→  errorFrame(type: AnthropicErrorType, message: string): string {
   710→    this.closed = true;
   711→    return sseFrame('error', { type: 'error', error: { type, message } });
   712→  }
   713→
   714→  get finalUsage(): TokenUsage | null {
   715→    return this.usage;
   716→  }
   717→}

   721→export function anthropicScopeFailure(key: AuthenticatedKey, scope: string): Response | null {
   722→  if (key.scopes.includes(scope)) return null;
   723→  return anthropicError(
   724→    403,
   725→    'permission_error',
   726→    `This API key is not permitted to use "${scope}". Edit the key's permissions in the EmorAI dashboard.`
   727→  );
   728→}
   729→
   730→export function anthropicModelAllowFailure(key: AuthenticatedKey, modelId: string): Response | null {
   731→  if (!key.allowedModels) return null;
   732→  const allowed = key.allowedModels.some((m) => m.toLowerCase() === modelId.toLowerCase());
   733→  if (allowed) return null;
   734→  return anthropicError(
   735→    403,
   736→    'permission_error',
   737→    `Model "${modelId}" is not on this key's allowed-models list. Allowed: ${key.allowedModels.join(', ')}.`
   738→  );
   739→}
