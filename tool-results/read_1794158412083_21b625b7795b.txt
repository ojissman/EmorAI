    27→    27→  | 'invalid_request_error'
    28→    28→  | 'authentication_error'

    38→    38→  message: string,
    39→    39→  headers?: Record<string, string>

    44→    44→/** Shapes a protocol-agnostic auth failure into the Anthropic envelope. */
    45→    45→export function anthropicAuthFailure(failure: AuthFailure): Response {
    46→    46→  const headers: Record<string, string> = {};
    47→    47→  if (failure.retryAfterSec) headers['retry-after'] = String(failure.retryAfterSec);
    48→    48→  return anthropicError(failure.status, failure.anthropicType, failure.message, headers);
    49→    49→}

    51→    51→/** Maps a provider (upstream) failure into the Anthropic error envelope. */
    52→    52→export function anthropicProviderError(code: string, message: string, status: number): Response {
    53→    53→  const type: AnthropicErrorType =
    54→    54→    code === 'rate_limited'
    55→    55→      ? 'rate_limit_error'
    56→    56→      : code === 'bad_request'
    57→    57→        ? 'invalid_request_error'
    58→    58→        : code === 'model_unavailable'
    59→    59→          ? 'api_error'
    60→    60→          : status >= 500
    61→    61→            ? 'overloaded_error'
    62→    62→            : 'api_error';
    63→    63→  const headers: Record<string, string> = {};
    64→    64→  if (code === 'rate_limited') headers['retry-after'] = '15';
    65→    65→  return anthropicError(status === 429 ? 429 : status >= 500 ? (status === 502 ? 529 : status) : status, type, message, headers);
    66→    66→}

   165→   165→export class ProtocolTranslationError extends Error {
   166→   166→  status: number;
   167→   167→  type: AnthropicErrorType;
   168→   168→  constructor(message: string, status = 400, type: AnthropicErrorType = 'invalid_request_error') {
   169→   169→    super(message);
   170→   170→    this.name = 'ProtocolTranslationError';
   171→   171→    this.status = status;
   172→   172→    this.type = type;
   173→   173→  }

   175→   175→
   176→   176→type RawBlock = AnthropicContentBlock;
   177→   177→
   178→   178→function joinText(blocks: RawBlock[]): string {
   179→   179→  return blocks
   180→   180→    .filter((b): b is Extract<RawBlock, { type: 'text' }> => b.type === 'text')
   181→   181→    .map((b) => b.text)
   182→   182→    .join('\n');
   183→   183→}
   184→   184→
   185→   185→function toolResultToText(block: Extract<RawBlock, { type: 'tool_result' }>): string {
   186→   186→  let text: string;
   187→   187→  if (typeof block.content === 'string') {
   188→   188→    text = block.content;
   189→   189→  } else if (Array.isArray(block.content)) {
   190→   190→    const parts = block.content.map((p) => {
   191→   191→      if (typeof (p as { text?: unknown }).text === 'string') return (p as { text: string }).text;
   192→   192→      if ((p as { type?: string }).type === 'image') {
   193→   193→        throw new ProtocolTranslationError(
   194→   194→          'Images inside tool_result blocks are not supported by the underlying provider. Return tool results as text.',
   195→   195→          400
   196→   196→        );
   197→   197→      }
   198→   198→      return '';
   199→   199→    });
   200→   200→    text = parts.filter(Boolean).join('\n');
   201→   201→  } else {
   202→   202→    text = '';
   203→   203→  }
   204→   204→  if (block.is_error) {
   205→   205→    text = text.length > 0 ? `[tool error] ${text}` : '[tool error]';
   206→   206→  }
   207→   207→  // The provider expects tool content as a string (JSON or plain text).
   208→   208→  return text.length > 0 ? text : '(empty tool result)';
   209→   209→}
   210→   210→
   211→   211→function imageSourceToPart(block: Extract<RawBlock, { type: 'image' }>): ProviderContentPart {
   212→   212→  const source = block.source as { type?: string; url?: string; media_type?: string; data?: string } | undefined;
   213→   213→  if (source?.type === 'url' && typeof source.url === 'string') {
   214→   214→    return { type: 'image_url', image_url: { url: source.url } };
   215→   215→  }
   216→   216→  if (source?.type === 'base64' && typeof source.data === 'string') {
   217→   217→    // Anthropic base64 image → OpenAI-style data URI
   218→   218→    const mediaType = source.media_type || 'image/png';
   219→   219→    return { type: 'image_url', image_url: { url: `data:${mediaType};base64,${source.data}` } };
   220→   220→  }
   221→   221→  // Malformed image block (passed the liberal fallback branch) — degrade to text
   222→   222→  return { type: 'text', text: '[invalid image block]' };
   223→   223→}
   224→   224→
   225→   225→function documentToText(block: Extract<RawBlock, { type: 'document' }>): string {
   226→   226→  const source = block.source as { type?: string; content?: string; media_type?: string } | undefined;
   227→   227→  if (source?.type === 'text' && typeof source.content === 'string') {
   228→   228→    return source.content;
   229→   229→  }
   230→   230→  throw new ProtocolTranslationError(
   231→   231→    'PDF document blocks are not supported by the underlying provider. Send document text inline or use a vision model with images.',
   232→   232→    400
   233→   233→  );
   234→   234→}
   235→   235→
   236→   236→/** Text of a message whose content is a string or an Anthropic block array. */
   237→   237→function contentToText(content: string | AnthropicContentBlock[]): string {
   238→   238→  if (typeof content === 'string') return content;
   239→   239→  return joinText(content);
   240→   240→}
   241→   241→
   242→   242→/**
   243→   243→ * Translates one normalized user/assistant message (string or block array)
   244→   244→ * into one or more provider messages. Tool results become individual `tool`
   245→   245→ * messages; text and images stay in `user`/`assistant` messages; thinking
   246→   246→ * blocks are accepted and dropped (the provider does not replay reasoning in
   247→   247→ * history).
   248→   248→ */
   249→   249→function translateMessage(msg: { role: 'user' | 'assistant'; content: string | AnthropicContentBlock[] }): ProviderMessage[] {
   250→   250→  if (typeof msg.content === 'string') {
   251→   251→    return [{ role: msg.role, content: msg.content }];
   252→   252→  }
   253→   253→  const blocks = msg.content;
   254→   254→
   255→   255→  if (msg.role === 'assistant') {
   256→   256→    const text = joinText(blocks);
   257→   257→    const toolUses = blocks.filter((b): b is Extract<RawBlock, { type: 'tool_use' }> => b.type === 'tool_use');
   258→   258→    if (toolUses.length > 0) {
   259→   259→      const toolCalls: ProviderToolCall[] = toolUses.map((u) => ({
   260→   260→        id: u.id,
   261→   261→        type: 'function',
   262→   262→        function: {
   263→   263→          name: u.name,
   264→   264→          arguments: JSON.stringify(u.input ?? {}),
   265→   265→        },
   266→   266→      }));
   267→   267→      return [{ role: 'assistant', content: text, tool_calls: toolCalls }];
   268→   268→    }
   269→   269→    return [{ role: 'assistant', content: text }];
   270→   270→  }
   271→   271→
   272→   272→  // user message: tool_result blocks must be answered first, then any text /
   273→   273→  // image content follows as a regular user message (ordering preserved).
   274→   274→  const out: ProviderMessage[] = [];
   275→   275→  for (const block of blocks) {
   276→   276→    if (block.type === 'tool_result') {
   277→   277→      out.push({ role: 'tool', content: toolResultToText(block), tool_call_id: block.tool_use_id });
   278→   278→    }
   279→   279→  }
   280→   280→  const rest = blocks.filter((b) => b.type !== 'tool_result');
   281→   281→  if (rest.length > 0) {
   282→   282→    const hasImages = rest.some((b) => b.type === 'image');
   283→   283→    if (hasImages) {
   284→   284→      const parts: ProviderContentPart[] = rest.map((b) => {
   285→   285→        if (b.type === 'text') return { type: 'text' as const, text: b.text };
   286→   286→        if (b.type === 'image') return imageSourceToPart(b);
   287→   287→        if (b.type === 'document') return { type: 'text' as const, text: documentToText(b) };
   288→   288→        // thinking blocks in user history are dropped
   289→   289→        return null;
   290→   290→      });
   291→   291→      const compact = parts.filter((p): p is ProviderContentPart => p !== null);
   292→   292→      if (compact.length > 0) out.push({ role: 'user', content: compact });
   293→   293→    } else {
   294→   294→      const texts = rest
   295→   295→        .map((b) => {
   296→   296→          if (b.type === 'text') return b.text;
   297→   297→          if (b.type === 'document') return documentToText(b);
   298→   298→          return null;
   299→   299→        })
   300→   300→        .filter((t): t is string => t !== null);
   301→   301→      if (texts.length > 0) out.push({ role: 'user', content: texts.join('\n') });
   302→   302→    }
   303→   303→  }
   304→   304→  return out;
   305→   305→}
   306→   306→
   307→   307→function translateSystem(system: AnthropicBody['system']): string | null {
   308→   308→  if (system === undefined) return null;
   309→   309→  if (typeof system === 'string') return system.length > 0 ? system : null;
   310→   310→  const texts = system
   311→   311→    .map((block) => (typeof block.text === 'string' ? block.text : ''))
   312→   312→    .filter(Boolean);
   313→   313→  return texts.length > 0 ? texts.join('\n\n') : null;
   314→   314→}
   315→   315→
   316→   316→function translateTools(tools: AnthropicBody['tools']): ProviderTool[] | undefined {
   317→   317→  if (!tools || tools.length === 0) return undefined;
   318→   318→  return tools.map((t) => ({
   319→   319→    type: 'function' as const,
   320→   320→    function: {
   321→   321→      name: t.name,
   322→   322→      description: t.description ?? '',
   323→   323→      parameters: t.input_schema ?? { type: 'object', properties: {} },
   324→   324→    },
   325→   325→  }));
   326→   326→}
   327→   327→
   328→   328→function translateToolChoice(choice: AnthropicBody['tool_choice']): unknown {
   329→   329→  if (!choice) return undefined;
   330→   330→  switch (choice.type) {
   331→   331→    case 'auto':
   332→   332→      return 'auto';
   333→   333→    case 'any':
   334→   334→      return 'required';
   335→   335→    case 'tool':
   336→   336→      return { type: 'function', function: { name: choice.name } };
   337→   337→  }
   338→   338→}

   352→   352→/** Full request translation, applying EmorAI model limits. */
   353→   353→export function anthropicToProvider(
   354→   354→  body: AnthropicBody,
   355→   355→  opts: { defaultMaxTokens: number; modelMaxOutputTokens: number | null }
   356→   356→): TranslatedRequest {
   357→   357→  if (body.messages.length === 0) {
   358→   358→    throw new ProtocolTranslationError('messages: at least one message is required.');
   359→   359→  }
   360→   360→
   361→   361→  // Liberal role handling: normalize each message role first, then translate.
   362→   362→  // system/developer roles merge into the system prompt; tool/function roles
   363→   363→  // become provider tool messages; OpenAI-style tool_calls on assistant
   364→   364→  // messages are honored. finalizeProviderMessages() then enforces the
   365→   365→  // upstream invariants (single leading system message, no consecutive
   366→   366→  // same-role messages, no empty messages).
   367→   367→  const providerMessages: ProviderMessage[] = [];
   368→   368→  const systemParts: string[] = [];
   369→   369→  const topSystem = translateSystem(body.system);
   370→   370→  if (topSystem) systemParts.push(topSystem);
   371→   371→
   372→   372→  for (const msg of body.messages) {
   373→   373→    const role = normalizeChatRole(msg.role);
   374→   374→    if (role === 'system') {
   375→   375→      const text = contentToText(msg.content);
   376→   376→      if (text.trim().length > 0) systemParts.push(text);
   377→   377→      continue;
   378→   378→    }
   379→   379→    if (role === 'tool') {
   380→   380→      // OpenAI-style tool result leaked into the Anthropic messages array
   381→   381→      const text = contentToText(msg.content);
   382→   382→      providerMessages.push({
   383→   383→        role: 'tool',
   384→   384→        content: text.trim().length > 0 ? text : '(empty tool result)',
   385→   385→        tool_call_id: msg.tool_call_id ?? msg.name ?? `call_${providerMessages.length}`,
   386→   386→      });
   387→   387→      continue;
   388→   388→    }
   389→   389→    if (role === 'assistant' && msg.tool_calls && msg.tool_calls.length > 0) {
   390→   390→      // OpenAI-style assistant tool_calls leak
   391→   391→      providerMessages.push({
   392→   392→        role: 'assistant',
   393→   393→        content: contentToText(msg.content),
   394→   394→        tool_calls: msg.tool_calls.map((tc, i) => ({
   395→   395→          id: tc.id && tc.id.length > 0 ? tc.id : `call_${Date.now().toString(36)}_${i}`,
   396→   396→          type: 'function' as const,
   397→   397→          function: { name: tc.function.name, arguments: tc.function.arguments ?? '{}' },
   398→   398→        })),
   399→   399→      });
   400→   400→      continue;
   401→   401→    }
   402→   402→    providerMessages.push(...translateMessage({ role, content: msg.content }));
   403→   403→  }
   404→   404→
   405→   405→  const messages = finalizeProviderMessages(
   406→   406→    systemParts.length > 0 ? [{ role: 'system', content: systemParts.join('\n\n') }, ...providerMessages] : providerMessages
   407→   407→  );
   408→   408→  if (messages.length === 0 || messages.every((m) => m.role === 'system')) {
   409→   409→    throw new ProtocolTranslationError('messages: the request contains no user or assistant content.');
   410→   410→  }
   411→   411→
   412→   412→  // Anthropic marks max_tokens optional in newer drafts but the classic API
   413→   413→  // requires it; we default to the model limit (friendlier than erroring).
   414→   414→  const requestedMax = body.max_tokens ?? opts.defaultMaxTokens;
   415→   415→  const maxTokens = Math.min(requestedMax, opts.modelMaxOutputTokens ?? requestedMax);
   416→   416→
   417→   417→  return {
   418→   418→    messages,
   419→   419→    tools: translateTools(body.tools),
   420→   420→    toolChoice: translateToolChoice(body.tool_choice),
   421→   421→    maxTokens,
   422→   422→    temperature: body.temperature,
   423→   423→    topP: body.top_p,
   424→   424→    topK: body.top_k,
   425→   425→    stop: body.stop_sequences && body.stop_sequences.length > 0 ? body.stop_sequences : undefined,
   426→   426→    thinkingEnabled: body.thinking?.type === 'enabled',
   427→   427→  };
   428→   428→}
   429→   429→
   430→   430→// ─── Token estimation ────────────────────────────────────────────────────────
   431→   431→
   432→   432→export function estimateInputTokens(messages: ProviderMessage[], system?: string | null): number {
   433→   433→  let chars = system?.length ?? 0;
   434→   434→  for (const m of messages) {
   435→   435→    if (typeof m.content === 'string') chars += m.content.length;
   436→   436→    else if (Array.isArray(m.content)) {
   437→   437→      for (const p of m.content) chars += p.text?.length ?? 0;
   438→   438→    }
   439→   439→    if (m.tool_calls) {
   440→   440→      for (const tc of m.tool_calls) chars += tc.function.arguments.length + tc.function.name.length;
   441→   441→    }
   442→   442→  }
   443→   443→  return Math.max(1, Math.ceil(chars / 4));
   444→   444→}

   448→   448→export type AnthropicStopReason = 'end_turn' | 'tool_use' | 'max_tokens' | 'stop_sequence' | 'refusal';
   449→   449→
   450→   450→export function toAnthropicStopReason(reason: string | null): AnthropicStopReason {
   451→   451→  switch (reason) {
   452→   452→    case 'tool_calls':
   453→   453→    case 'function_call':
   454→   454→      return 'tool_use';
   455→   455→    case 'length':
   456→   456→    case 'max_tokens':
   457→   457→      return 'max_tokens';
   458→   458→    case 'content_filter':
   459→   459→      return 'refusal';
   460→   460→    default:
   461→   461→      return 'end_turn';
   462→   462→  }
   463→   463→}

   465→   465→export type AnthropicContentOut =
   466→   466→  | { type: 'text'; text: string }
   467→   467→  | { type: 'tool_use'; id: string; name: string; input: Record<string, unknown> };
   468→   468→
   469→   469→function safeParseArguments(args: string): Record<string, unknown> {
   470→   470→  try {
   471→   471→    const parsed = JSON.parse(args || '{}');
   472→   472→    return typeof parsed === 'object' && parsed !== null && !Array.isArray(parsed)
   473→   473→      ? (parsed as Record<string, unknown>)
   474→   474→      : { value: parsed };
   475→   475→  } catch {
   476→   476→    // Malformed arguments from the provider — surface raw text honestly.
   477→   477→    return { _raw: args };
   478→   478→  }
   479→   479→}

   481→   481→/** Builds the non-streaming Anthropic message response. */
   482→   482→export function buildAnthropicMessage(params: {
   483→   483→  id: string;
   484→   484→  requestedModel: string;
   485→   485→  servedModel: string | null;
   486→   486→  content: string;
   487→   487→  toolCalls: ProviderToolCall[];
   488→   488→  finishReason: string | null;
   489→   489→  usage: TokenUsage | null;
   490→   490→}): Record<string, unknown> {
   491→   491→  const content: AnthropicContentOut[] = [];
   492→   492→  if (params.content && params.content.length > 0) {
   493→   493→    content.push({ type: 'text', text: params.content });
   494→   494→  }
   495→   495→  for (const tc of params.toolCalls) {
   496→   496→    content.push({
   497→   497→      type: 'tool_use',
   498→   498→      id: tc.id,
   499→   499→      name: tc.function.name,
   500→   500→      input: safeParseArguments(tc.function.arguments),
   501→   501→    });
   502→   502→  }
   503→   503→  if (content.length === 0) {
   504→   504→    content.push({ type: 'text', text: '' });
   505→   505→  }
   506→   506→  return {
   507→   507→    id: params.id,
   508→   508→    type: 'message',
   509→   509→    role: 'assistant',
   510→   510→    model: params.requestedModel,
   511→   511→    content,
   512→   512→    stop_reason: toAnthropicStopReason(params.finishReason),
   513→   513→    stop_sequence: null,
   514→   514→    usage: {
   515→   515→      input_tokens: params.usage?.promptTokens ?? 0,
   516→   516→      output_tokens: params.usage?.completionTokens ?? 0,
   517→   517→    },
   518→   518→    // EmorAI transparency extension (ignored by Anthropic SDKs)
   519→   519→    emorai_served_model: params.servedModel ?? params.requestedModel,
   520→   520→  };
   521→   521→}

   523→   523→// ─── Streaming translation: provider events → Anthropic SSE frames ───────────
   524→   524→
   525→   525→function sseFrame(event: string, data: unknown): string {
   526→   526→  return `event: ${event}\ndata: ${JSON.stringify(data)}\n\n`;
   527→   527→}
   528→   528→
   529→   529→type OpenBlock =
   530→   530→  | { kind: 'text'; index: number }
   531→   531→  | { kind: 'tool_use'; index: number; id: string; name: string };
   532→   532→
   533→   533→/**
   534→   534→ * Stateful translator that consumes normalized provider stream events and
   535→   535→ * produces the Anthropic SSE event sequence. Faithful to the protocol:
   536→   536→ *   message_start → ping → (content_block_start → deltas → content_block_stop)*
   537→   537→ *   → message_delta(stop_reason, usage) → message_stop
   538→   538→ * Tool-call arguments stream as input_json_delta chunks.
   539→   539→ */
   540→   540→export class AnthropicStreamTranslator {
   541→   541→  private readonly messageId: string;
   542→   542→  private readonly requestedModel: string;
   543→   543→  private blockCount = 0;
   544→   544→  private open: OpenBlock | null = null;
   545→   545→  private outputChars = 0;
   546→   546→  private usage: TokenUsage | null = null;
   547→   547→  private inputTokens: number;
   548→   548→  private closed = false;
   549→   549→
   550→   550→  constructor(params: { messageId: string; requestedModel: string; estimatedInputTokens: number }) {
   551→   551→    this.messageId = params.messageId;
   552→   552→    this.requestedModel = params.requestedModel;
   553→   553→    this.inputTokens = params.estimatedInputTokens;
   554→   554→  }
   555→   555→
   556→   556→  private nextIndex(): number {
   557→   557→    return this.blockCount++;
   558→   558→  }
   559→   559→
   560→   560→  private closeOpenBlock(): string | null {
   561→   561→    if (!this.open) return null;
   562→   562→    const frame = sseFrame('content_block_stop', { type: 'content_block_stop', index: this.open.index });
   563→   563→    this.open = null;
   564→   564→    return frame;
   565→   565→  }
   566→   566→
   567→   567→  /** Initial frames: message_start + ping. */
   568→   568→  start(): string {
   569→   569→    return (
   570→   570→      sseFrame('message_start', {
   571→   571→        type: 'message_start',
   572→   572→        message: {
   573→   573→          id: this.messageId,
   574→   574→          type: 'message',
   575→   575→          role: 'assistant',
   576→   576→          model: this.requestedModel,
   577→   577→          content: [],
   578→   578→          stop_reason: null,
   579→   579→          stop_sequence: null,
   580→   580→          usage: { input_tokens: this.inputTokens, output_tokens: 1 },
   581→   581→        },
   582→   582→      }) + sseFrame('ping', { type: 'ping' })
   583→   583→    );
   584→   584→  }
   585→   585→
   586→   586→  /** Feed one normalized provider event; returns SSE frames to write. */
   587→   587→  onEvent(ev: StreamEvent): string {
   588→   588→    if (this.closed) return '';
   589→   589→    switch (ev.type) {
   590→   590→      case 'delta': {
   591→   591→        this.outputChars += ev.content.length;
   592→   592→        if (!this.open || this.open.kind !== 'text') {
   593→   593→          const closeFrame = this.closeOpenBlock();
   594→   594→          const index = this.nextIndex();
   595→   595→          this.open = { kind: 'text', index };
   596→   596→          const startFrame = sseFrame('content_block_start', {
   597→   597→            type: 'content_block_start',
   598→   598→            index,
   599→   599→            content_block: { type: 'text', text: '' },
   600→   600→          });
   601→   601→          return (
   602→   602→            (closeFrame ?? '') +
   603→   603→            startFrame +
   604→   604→            sseFrame('content_block_delta', {
   605→   605→              type: 'content_block_delta',
   606→   606→              index,
   607→   607→              delta: { type: 'text_delta', text: ev.content },
   608→   608→            })
   609→   609→          );
   610→   610→        }
   611→   611→        return sseFrame('content_block_delta', {
   612→   612→          type: 'content_block_delta',
   613→   613→          index: this.open.index,
   614→   614→          delta: { type: 'text_delta', text: ev.content },
   615→   615→        });
   616→   616→      }
   617→   617→      case 'reasoning': {
   618→   618→        // GLM reasoning trace: Anthropic surfaces extended thinking as
   619→   619→        // thinking blocks. We stream them as thinking blocks (signature is
   620→   620→        // empty; clients tolerate unsigned thinking on custom gateways).
   621→   621→        if (!this.open || this.open.kind !== 'text') {
   622→   622→          // reasoning arrives before text typically — open a thinking block
   623→   623→          // only if nothing is open; otherwise ignore (kept simple & robust).
   624→   624→          if (this.open) return '';
   625→   625→          const index = this.nextIndex();
   626→   626→          this.open = { kind: 'text', index };
   627→   627→          // Represent reasoning as a thinking-typed content block
   628→   628→          return (
   629→   629→            sseFrame('content_block_start', {
   630→   630→              type: 'content_block_start',
   631→   631→              index,
   632→   632→              content_block: { type: 'thinking', thinking: '', signature: '' },
   633→   633→            }) +
   634→   634→            sseFrame('content_block_delta', {
   635→   635→              type: 'content_block_delta',
   636→   636→              index,
   637→   637→              delta: { type: 'thinking_delta', thinking: ev.content },
   638→   638→            })
   639→   639→          );
   640→   640→        }
   641→   641→        return '';
   642→   642→      }
   643→   643→      case 'tool-call-delta': {
   644→   644→        return this.onToolCallDelta(ev.call);
   645→   645→      }
   646→   646→      case 'served-model': {
   647→   647→        // No Anthropic frame for this; recorded by the caller.
   648→   648→        return '';
   649→   649→      }
   650→   650→      case 'usage': {
   651→   651→        this.usage = ev.usage;
   652→   652→        return '';
   653→   653→      }
   654→   654→      case 'finish': {
   655→   655→        // handled in finish(); nothing to emit now
   656→   656→        return '';
   657→   657→      }
   658→   658→    }
   659→   659→  }
   660→   660→
   661→   661→  private onToolCallDelta(call: ToolCallDelta): string {
   662→   662→    let frames = '';
   663→   663→    const sameToolOpen =
   664→   664→      this.open && this.open.kind === 'tool_use' && this.open.id === (call.id ?? this.open.id);
   665→   665→    if (!sameToolOpen) {
   666→   666→      frames += this.closeOpenBlock() ?? '';
   667→   667→      const index = this.nextIndex();
   668→   668→      const id = call.id ?? `toolu_${Math.random().toString(36).slice(2, 12)}`;
   669→   669→      const name = call.name ?? '';
   670→   670→      this.open = { kind: 'tool_use', index, id, name };
   671→   671→      frames += sseFrame('content_block_start', {
   672→   672→        type: 'content_block_start',
   673→   673→        index,
   674→   674→        content_block: { type: 'tool_use', id, name, input: {} },
   675→   675→      });
   676→   676→    }
   677→   677→    if (call.argumentsDelta && call.argumentsDelta.length > 0) {
   678→   678→      this.outputChars += call.argumentsDelta.length;
   679→   679→      frames += sseFrame('content_block_delta', {
   680→   680→        type: 'content_block_delta',
   681→   681→        index: this.open!.index,
   682→   682→        delta: { type: 'input_json_delta', partial_json: call.argumentsDelta },
   683→   683→      });
   684→   684→    }
   685→   685→    return frames;
   686→   686→  }
   687→   687→
   688→   688→  /** Terminal frames: close block, message_delta with stop_reason + usage, message_stop. */
   689→   689→  finish(finishReason: string | null): string {
   690→   690→    if (this.closed) return '';
   691→   691→    this.closed = true;
   692→   692→    const closeFrame = this.closeOpenBlock() ?? '';
   693→   693→    const stopReason = toAnthropicStopReason(finishReason);
   694→   694→    const outputTokens =
   695→   695→      this.usage?.completionTokens ?? Math.max(1, Math.ceil(this.outputChars / 4));
   696→   696→    if (this.usage) this.inputTokens = this.usage.promptTokens;
   697→   697→    return (
   698→   698→      closeFrame +
   699→   699→      sseFrame('message_delta', {
   700→   700→        type: 'message_delta',
   701→   701→        delta: { stop_reason: stopReason, stop_sequence: null },
   702→   702→        usage: { output_tokens: outputTokens },
   703→   703→      }) +
   704→   704→      sseFrame('message_stop', { type: 'message_stop' })
   705→   705→    );
   706→   706→  }
   707→   707→
   708→   708→  /** Mid-stream error frame (Anthropic error event). */
   709→   709→  errorFrame(type: AnthropicErrorType, message: string): string {
   710→   710→    this.closed = true;
   711→   711→    return sseFrame('error', { type: 'error', error: { type, message } });
   712→   712→  }
   713→   713→
   714→   714→  get finalUsage(): TokenUsage | null {
   715→   715→    return this.usage;
   716→   716→  }
   717→   717→}

   730→   730→export function anthropicModelAllowFailure(key: AuthenticatedKey, modelId: string): Response | null {
   731→   731→  if (!key.allowedModels) return null;
   732→   732→  const allowed = key.allowedModels.some((m) => m.toLowerCase() === modelId.toLowerCase());
   733→   733→  if (allowed) return null;
   734→   734→  return anthropicError(
   735→   735→    403,
   736→   736→    'permission_error',
   737→   737→    `Model "${modelId}" is not on this key's allowed-models list. Allowed: ${key.allowedModels.join(', ')}.`
   738→   738→  );
   739→   739→}
